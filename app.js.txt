document.addEventListener('DOMContentLoaded', () => {
  const form = document.getElementById('appForm');
  const input = document.getElementById('itemInput');
  const itemList = document.getElementById('itemList');

  // Load saved items from localStorage
  const savedItems = JSON.parse(localStorage.getItem('app_items')) || [];
  savedItems.forEach(text => createItemElement(text));

  // Form submission handler
  form.addEventListener('submit', (e) => {
    e.preventDefault();
    const text = input.value.trim();
    if (text) {
      createItemElement(text);
      saveItemToStorage(text);
      input.value = '';
    }
  });

  // Function to create DOM elements
  function createItemElement(text) {
    const itemCard = document.createElement('div');
    itemCard.className = 'flex items-center justify-between p-3.5 bg-slate-50 border border-slate-200 rounded-xl shadow-sm animate-fade-in';
    
    itemCard.innerHTML = `
      <span class="text-sm font-medium text-slate-700">${escapeHtml(text)}</span>
      <button class="delete-btn text-slate-400 hover:text-red-500 text-xs font-semibold">Delete</button>
    `;

    // Delete item functionality
    itemCard.querySelector('.delete-btn').addEventListener('click', () => {
      itemCard.remove();
      removeItemFromStorage(text);
    });

    itemList.prepend(itemCard);
  }

  // Local Storage Helpers
  function saveItemToStorage(text) {
    const items = JSON.parse(localStorage.getItem('app_items')) || [];
    items.push(text);
    localStorage.setItem('app_items', JSON.stringify(items));
  }

  function removeItemFromStorage(text) {
    let items = JSON.parse(localStorage.getItem('app_items')) || [];
    items = items.filter(item => item !== text);
    localStorage.setItem('app_items', JSON.stringify(items));
  }

  function escapeHtml(str) {
    return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
  }
});
